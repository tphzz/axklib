# Private, pinned XML parser; no host dependency or configure-time download.
include(TestBigEndian)
test_big_endian(expat_big_endian)
if(expat_big_endian)
    set(EXPAT_BYTEORDER 4321)
else()
    set(EXPAT_BYTEORDER 1234)
endif()
configure_file(expat_config.h.in expat_config.h @ONLY)
set(expat_sources ../../vendor/expat/xmlparse.c ../../vendor/expat/xmlrole.c ../../vendor/expat/xmltok.c)
if(WIN32)
    list(APPEND expat_sources ../../vendor/expat/random_rand_s.c)
else()
    list(APPEND expat_sources ../../vendor/expat/random_dev_urandom.c)
endif()
add_library(rawdisk_expat STATIC ${expat_sources})
target_include_directories(rawdisk_expat SYSTEM PUBLIC ../../vendor/expat)
target_include_directories(rawdisk_expat PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(rawdisk_expat PUBLIC XML_STATIC PRIVATE XML_BUILDING_EXPAT)
set_target_properties(rawdisk_expat PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON
    POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
